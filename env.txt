# GigaCode Proxy (Node.js) — Configuration (ПРИМЕР)
# =============================================================================
#   ЭТО ШАБЛОН. Скопируйте его в env.txt и правьте уже копию:
#
#       Windows:  copy env.example env.txt
#       Linux/Mac: cp env.example env.txt
#
#   Сам env.example прокси НЕ читает. env.txt в .gitignore — ваш логин
#   и токены не попадут в репозиторий.
# =============================================================================
# Файл env.txt читается автоматически — переименовывать в .env НЕ нужно.
# Прокси ищет по порядку: ./env.txt, ./.env, <папка server.mjs>/env.txt,
# <папка server.mjs>/.env. Реальные переменные окружения имеют приоритет.
#
# Синтаксис: KEY=value, одна пара на строку. Кавычки не обязательны,
# строки с # игнорируются. Значения НЕ подставляются друг в друга.
# =============================================================================


# -----------------------------------------------------------------------------
# Авторизация (ОБЯЗАТЕЛЬНО)
# -----------------------------------------------------------------------------
# Логин SberID. При первом запуске откроется браузер для входа,
# затем токен кэшируется на диск и логин больше не потребуется.
GIGACODE_USERNAME=


# -----------------------------------------------------------------------------
# Сервер прокси
# -----------------------------------------------------------------------------
# 127.0.0.1 — доступ только с этой машины (безопасно).
# 0.0.0.0    — доступ из локальной сети (нужно для Docker/WSL/другого ПК).
PROXY_HOST=127.0.0.1
PROXY_PORT=8000

# Модель по умолчанию, если редактор не прислал имя.
MODEL_NAME=CodeChat


# -----------------------------------------------------------------------------
# Генерация
# -----------------------------------------------------------------------------
# Потолок вывода, когда клиент не прислал свой max_tokens.
# 0 = прокси ничего не навязывает, длину решает редактор (рекомендуется).
GIGACODE_DEFAULT_MAX_TOKENS=0

# ВАЖНО: temperature / top_p / frequency_penalty / presence_penalty и любые
# другие поля (seed, logit_bias, top_k, reasoning_effort, вендорские
# расширения) прокси НЕ трогает — они уходят наверх как есть.
# Отдельных переменных для них здесь нет и не нужно: настраивай в редакторе.


# -----------------------------------------------------------------------------
# Passthrough: последние преобразования запроса
# -----------------------------------------------------------------------------
# Имя модели: 1 = слать как есть, 0 = резолвить алиасы и откатываться
# на MODEL_NAME, если имя незнакомое (спасает пресеты вроде "gpt-4o").
GIGACODE_PASSTHROUGH_MODEL=1

# GigaCode не принимает роль "system" — заменяем её.
# Пустое значение = не трогать роль (GigaCode, скорее всего, ответит ошибкой).
GIGACODE_SYSTEM_ROLE=user

# Orphan tool-сообщения в начале истории вызывают 422.
# 0 = вырезать их (безопасно), 1 = не трогать, чистый passthrough.
GIGACODE_KEEP_ORPHAN_TOOLS=1

# GigaCode текстовый: 1 = схлопывать мультимодальный контент в текст
# (картинки станут пометкой), 0 = слать структуру как есть.
GIGACODE_FLATTEN_MULTIMODAL=1


# -----------------------------------------------------------------------------
# Прогресс на агентских шагах
# -----------------------------------------------------------------------------
# Агентские шаги (вызовы инструментов) редактор шлёт с stream=false и ждёт
# ответ целиком — иногда десятки секунд полной тишины, будто всё зависло.
# 1 = запрашивать у GigaCode поток даже для таких запросов и собирать ответ
#     на стороне прокси. Клиент получает тот же самый JSON, но в консоли
#     видно живой прогресс ([WAIT] 5s chars=120 ...), а соединение не простаивает.
# 0 = старое поведение (один блокирующий запрос).
GIGACODE_STREAM_NONSTREAM=0


# -----------------------------------------------------------------------------
# Рассуждения модели (chain-of-thought)
# -----------------------------------------------------------------------------
# Пока модель "думает", обычный content пуст — кажется, что она зависла.
# Этот режим показывает ход мыслей:
#   tags  - вшить в ответ как <think>...</think> (видно в ЛЮБОМ редакторе,
#           многие сворачивают такой блок автоматически)
#   field - отдельное поле reasoning_content по стандарту OpenAI
#           (Cline, Roo, Open WebUI рисуют его сворачиваемым блоком;
#            редакторы без поддержки просто ничего не покажут)
#   off   - выбрасывать рассуждения
#
# ВАЖНО: если GigaCode вообще не присылает мысли отдельным полем,
# показывать нечего ни в одном режиме. Проверяется через
# GIGACODE_DEBUG_RAW=1 (см. раздел "Отладка" ниже): в логе появятся
# строки [RAW] и [FIELD] с реальными полями ответа.
# CodeChat всегда присылает reasoning_content=null, поэтому по умолчанию off.
GIGACODE_SHOW_REASONING=off

# Пинг-комментарий в SSE каждые N мс, пока нет токенов, — не даёт редактору
# считать соединение мёртвым при долгих паузах. 0 = выключить.
GIGACODE_KEEPALIVE_MS=15000


# -----------------------------------------------------------------------------
# Совместимость с редакторами
# -----------------------------------------------------------------------------
# Прокси и так принимает любые префиксы путей (/v1, /api/v1, /openai/v1),
# отдаёт список моделей с алиасами и чинит форму SSE-чанков.
# Если редактор жёстко требует своё имя модели, допиши его сюда через запятую —
# оно появится в /v1/models, а запрос уйдёт на CodeChat.
# GIGACODE_EXTRA_MODELS=gpt-4o,claude-3-5-sonnet-20241022


# -----------------------------------------------------------------------------
# Кэш токена
# -----------------------------------------------------------------------------
# По умолчанию: <домашняя папка>/.gigacode_proxy_tokens.json
# Device code:  <домашняя папка>/.gigacode_device_code
# Удали эти файлы, чтобы принудительно перелогиниться.
# GIGACODE_TOKEN_FILE=C:\Users\Admin\.gigacode_proxy_tokens.json

# Постоянный идентификатор устройства, 64 hex-символа.
# Если не задан — генерируется при первом запуске и сохраняется на диск.
# GIGACODE_DEVICE_CODE=0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef


# -----------------------------------------------------------------------------
# Отладка
# -----------------------------------------------------------------------------
# 1 = логировать каждый SSE-кусок от GigaCode. Шумно и замедляет вывод,
# но помогает диагностировать дубли текста и обрывы tool_calls.
GIGACODE_DEBUG_CHUNKS=1

# 1 = печатать КАЖДУЮ сырую SSE-строку от GigaCode ([RAW] ...).
# Включи, если рассуждения не видны: в логе будет точно видно,
# присылает ли сервер поле с мыслями и как оно называется.
GIGACODE_DEBUG_RAW=1

# ПОЛНАЯ ТРАССИРОВКА (включай, когда что-то не работает).
# 1 = писать каждый шаг и в консоль, и в файл trace.log рядом с server.mjs:
#   [IN] / [IN-BODY]   что прислал редактор
#   [UP-REQ]/[UP-RES]  что ушло в GigaCode и что он ответил
#   [RAW]              каждая сырая SSE-строка от GigaCode
#   [TC-FRAG]          каждый кусок аргументов инструмента по мере прихода
#   [TC-IN]/[TC-CAND]/[TC-PICKED]/[TC-OUT]  как собирался итоговый вызов
#   [OUT-CHUNK]/[OUT-JSON]  что прокси отдал редактору
GIGACODE_TRACE=1

# Куда писать трассировку. По умолчанию trace.log рядом с server.mjs.
# Можно указать файл и при GIGACODE_TRACE=0 — тогда консоль остаётся чистой,
# а полный лог всё равно пишется на диск.
# GIGACODE_TRACE_FILE=C:\Alarms\js\gigacode-proxy\trace.log

# Обрезка длинных значений в трассировке (символов). 0 не бывает — ставь много.
GIGACODE_TRACE_MAX_CHARS=0


# =============================================================================
# Эндпоинты и таймауты (трогать обычно не нужно)
# =============================================================================
# GIGACODE_BASE_URL=https://gigacode.ru
# GIGACODE_AUTH_URL=https://gigacode.ru/api/v2/auth/plugin
# GIGACODE_CHAT_URL=https://gigacode.ru/api/v2/chat/completions
# GIGACODE_COMPLETIONS_URL=https://gigacode.ru/api/v2/completions
# GIGACODE_RESPONSES_URL=https://gigacode.ru/api/v2/responses
# GIGACODE_CALLBACK_PATH=/gigaCodeAuthCallback

# Таймауты в секундах. REQUEST_TIMEOUT стоит поднять, если длинные
# ответы обрываются на середине.
# CONNECT_TIMEOUT=7
# REQUEST_TIMEOUT=50


# =============================================================================
# Переменных из Python-версии больше НЕТ — они там ни на что не влияли:
#   GIGACODE_MAX_HISTORY_MESSAGES   история не обрезается
#   GIGACODE_MAX_TOOL_RESULT_CHARS  результаты инструментов не режутся
#   GIGACODE_MAX_PROMPT_CHARS       размер промпта не ограничивается
#   GIGACODE_MIN_PROMPT_CHARS       сжатия промпта нет
#   GIGACODE_MAX_RETRIES            ретраи 401/429/5xx зашиты в код
#   CALLBACK_HOST / CALLBACK_PORT   порт колбэка выбирается свободный
#   SOCKET_TIMEOUT                  не используется fetch'ем
# =============================================================================


# =============================================================================
# Telegram-бот (bot.mjs) — читает эти же файлы
# =============================================================================
# Токен от @BotFather: /newbot → скопировать строку вида 123456789:AA...
TELEGRAM_BOT_TOKEN=

# Кто может пользоваться ботом. Пока пусто и BOT_ALLOW_ALL=0 — бот на любое
# сообщение отвечает: «Бот закрыт. Ваш user id: …» — впишите этот id сюда.
BOT_ALLOWED_USERS=1
# 1 = пустить всех подряд (НЕ рекомендуется: любой будет тратить вашу квоту).
BOT_ALLOW_ALL=

# 1 = бот сам поднимает server.mjs, если тот ещё не запущен (файл рядом).
BOT_AUTOSTART_PROXY=1
# 1 = разрешить работу в группах (бот отвечает на упоминание или реплай).
BOT_ALLOW_GROUPS=1

# Контекст диалога: сколько помнить и сколько символов держать в истории.
BOT_HISTORY_MAX_MESSAGES=40
BOT_HISTORY_CHAR_BUDGET=60000

# Показывать размышления модели отдельным сворачиваемым блоком (1/0).
BOT_SHOW_REASONING=0
# Печатать ответ по мере генерации (1) или после полной готовности (0).
BOT_STREAM=1

# Логи бота: дублировать в файл (пусто = только консоль).
BOT_LOG_FILE=
# Сколько секунд молчания прокси считать обрывом соединения.
BOT_IDLE_TIMEOUT_MS=180000
